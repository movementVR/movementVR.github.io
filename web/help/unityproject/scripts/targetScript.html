Documentation: targetScript.cs

1. Overview

The targetScript handles logic for determining whether the plate has been successfully placed within the target area in MovementVR. It monitors collision timing, tilt limits, and visual feedback to trigger success or failure based on task criteria. This script is critical to trial validation and feedback control.

2. Functional Blocks

Setup

  • startup(): Retrieves references to the trial controller, plate script, and plate trigger collider.

Collision Detection

  • OnTriggerEnter(): Called when the plate enters the target; stores entry time and activates success color.
  • OnTriggerStay(): If enough time has passed and the plate is stable, triggers success. If tilted too much, triggers failure.
  • OnTriggerExit(): Triggers failure if the plate leaves the target prematurely.

Visual Feedback

  • setMeshSuccess(): Changes mesh color based on success or failure.
  • resetTarget(): Resets the mesh state and internal flags before a new trial.

3. Full Technical Details

Fields

// References trialScript trialScriptVar; plateScript plateScriptVar; Collider plateTrigger; MeshRenderer targetMesh; // Timing System.DateTime collisionTime; float timeInTargetForSuccess; // Tilt thresholds float maxTiltForSuccessX; float maxTiltForSuccessY; float maxTiltForSuccessZ; // Colors float targetMeshColorSuccessR, targetMeshColorSuccessG, targetMeshColorSuccessB, targetMeshColorSuccessAlpha; float targetMeshColorUnsuccessR, targetMeshColorUnsuccessG, targetMeshColorUnsuccessB, targetMeshColorUnsuccessAlpha; // State public bool plateInTarget;

startup()

public void startup() { trialScriptVar = transform.Find("/Task").GetComponent<trialScript>(); plateScriptVar = transform.Find("/Task/PlateAndGrape/Plate").GetComponent<plateScript>(); plateTrigger = transform.Find("/Task/PlateAndGrape/Plate/plateTrigger").GetComponent<Collider>(); }

OnTriggerEnter()

private void OnTriggerEnter(Collider other) { if (other == plateTrigger) { collisionTime = System.DateTime.Now; setMeshSuccess(true); plateInTarget = true; } }

OnTriggerStay()

private void OnTriggerStay(Collider other) { if (other == plateTrigger) { if ((System.DateTime.Now - collisionTime).TotalSeconds > timeInTargetForSuccess) { trialScriptVar.trialStageTrigger(1); // success } else if (plateScriptVar.plateTooTilted(maxTiltForSuccessX, maxTiltForSuccessY, maxTiltForSuccessZ)) { trialScriptVar.trialStageTrigger(-1); // fail: tilt setMeshSuccess(false); } } }

OnTriggerExit()

private void OnTriggerExit(Collider other) { if (other == plateTrigger) { trialScriptVar.trialStageTrigger(-2); // fail: left target setMeshSuccess(false); } }

Visual Feedback Methods

public void setMeshSuccess(bool success) { Color c = success ? new Color(targetMeshColorSuccessR, targetMeshColorSuccessG, targetMeshColorSuccessB, targetMeshColorSuccessAlpha) : new Color(targetMeshColorUnsuccessR, targetMeshColorUnsuccessG, targetMeshColorUnsuccessB, targetMeshColorUnsuccessAlpha); targetMesh.material.color = c; } public void resetTarget() { targetMesh = GetComponentInChildren<MeshRenderer>(); setMeshSuccess(false); plateInTarget = false; }
