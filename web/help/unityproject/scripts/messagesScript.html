Documentation: messagesScript.cs

1. Overview

messagesScript manages on-screen instructional and feedback messages shown to the user during a trial in a VR motor learning task. These include messages for stages like origin, contact, lift, and end-of-trial feedback (e.g., success or various types of failure). The class allows toggling messages by trial stage and dynamically updates trial numbers.

2. Functional Blocks

Text and Display Control

  • Stores text content for all messages as private string fields.
  • Flags for whether each stage should display a message.
  • Controls message visibility through messageDisplay(), using reflection to access the relevant text field.

Trial Stage Transitions

  • changeTrialStage() determines which messages to show or hide at each stage (1 to 6).
  • Uses the info argument to differentiate between trial failure reasons and end-of-paradigm conditions.

Integration with Trial Script

  • References trialScript and uses trialStageTrigger() to progress when messages are skipped or complete.

messageDisplay()

public void messageDisplay(string msg, bool state) { if (state) { transform.Find(msg + "/text").GetComponent<TMPro.TextMeshPro>().text = (string)GetType().GetField(msg + "Text", BindingFlags.Instance | BindingFlags.NonPublic).GetValue(this); } transform.Find(msg).gameObject.SetActive(state); }

changeTrialStage()

public void changeTrialStage(int stage, float info) { switch (stage) { case 1: messageDisplay("break", false); if (stage1_displayMessageFlag) messageDisplay("base", true); if (info != -1) { trialNumberText = trialNumberAuto ? trialNumberBaseText + info.ToString() : trialNumberBaseText; messageDisplay("trialNumber", true); } else { messageDisplay("end", true); } break; case 2: if (stage1_displayMessageFlag) messageDisplay("base", false); if (stage2_displayMessageFlag) messageDisplay("origin", true); break; case 3: if (stage2_displayMessageFlag) messageDisplay("origin", false); if (stage3_displayMessageFlag) messageDisplay("contact", true); break; case 4: if (stage3_displayMessageFlag) messageDisplay("contact", false); if (stage4_displayMessageFlag) messageDisplay("lift", true); break; case 5: if (stage4_displayMessageFlag) messageDisplay("lift", false); if (stage5_displayMessageFlag) { feedbackMessage = (info < 0) ? "fail" + (-info).ToString() : "success"; string text = (string)GetType().GetField(feedbackMessage + "Text", BindingFlags.Instance | BindingFlags.NonPublic).GetValue(this); if (!string.IsNullOrEmpty(text)) { messageDisplay(feedbackMessage, true); } else { trialScriptVar.trialStageTrigger(3); // Skip feedback display } } break; case 6: if (stage5_displayMessageFlag) messageDisplay(feedbackMessage, false); if (stage6_displayMessageFlag && info > 0) messageDisplay("break", true); break; } }
