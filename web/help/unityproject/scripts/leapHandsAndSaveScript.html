Documentation: leapHandsAndSaveScript.cs

1. Overview

leapHandsAndSaveScript extends handsAndSaveScript to support Leap Motion-specific hand tracking and recording. It connects Unity hand models and colliders with Leap’s rigging and runtime control, tracks their transform and state, and saves relevant bone and collider data during runtime for analysis. It also applies gain perturbation offsets to hands and fingers in real time.

2. Functional Blocks

Leap Rig Setup

  • Finds Leap Motion head and hand model transforms.
  • Initializes references to both mesh (visual) and rigid (physics) hands and fingers.
  • Populates collider arrays for transform and shape recording.

Tracking and Activation

  • leftNotTracking() / rightNotTracking(): Returns whether the mesh hand is inactive.
  • setHandGameobjectActivity(): Enables or disables Leap hand groups depending on active flags.

Data Logging Extensions

  • additionalFrameSave(): Saves additional data about hand collider shapes (capsules and boxes).
  • additionalInfoSave(): Reserved for saving extra metadata (empty by default).

Perturbation Application

  • applyLeftPerturbation() and applyRightPerturbation(): Apply gain offsets to the entire hand and all five fingers for both visual and physical models.

rigSetup()

protected override void rigSetup() { Debug.Log("AABBleap"); headsetTransform = transform.Find("/LeapMotion/HeadsetCamera"); handsParentObject = transform.Find("/LeapMotion/HandModels"); // Left hand leftHandMeshScript = handsParentObject.Find("LeftHandMesh").GetComponent<Leap.Unity.RiggedHand>(); leftFingersMeshScripts = handsParentObject.Find("LeftHandMesh").GetComponentsInChildren<Leap.Unity.RiggedFinger>(true); leftHandColliderScript = handsParentObject.Find("LeftHandRigidBody").GetComponent<Leap.Unity.RigidHand>(); leftFingersColliderScripts = handsParentObject.Find("LeftHandRigidBody").GetComponentsInChildren<Leap.Unity.RigidFinger>(true); // Right hand rightHandMeshScript = handsParentObject.Find("RightHandMesh").GetComponent<Leap.Unity.RiggedHand>(); rightFingersMeshScripts = handsParentObject.Find("RightHandMesh").GetComponentsInChildren<Leap.Unity.RiggedFinger>(true); rightHandColliderScript = handsParentObject.Find("RightHandRigidBody").GetComponent<Leap.Unity.RigidHand>(); rightFingersColliderScripts = handsParentObject.Find("RightHandRigidBody").GetComponentsInChildren<Leap.Unity.RigidFinger>(true); // Wrists leftWrist = handsParentObject.Find("LeftHandRigidBody/wrist"); rightWrist = handsParentObject.Find("RightHandRigidBody/wrist"); // All hand colliders allCollidersInHands = handsParentObject.GetComponentsInChildren<Collider>(true); }

Hand Tracking Checks

protected override bool leftNotTracking() { return !leftHandMeshScript.isActiveAndEnabled; } protected override bool rightNotTracking() { return !rightHandMeshScript.isActiveAndEnabled; }

setHandGameobjectActivity()

protected override void setHandGameobjectActivity() { var manager = handsParentObject.GetComponent<Leap.Unity.HandModelManager>(); if (leftHandActiveFlag) { manager.EnableGroup("LeftMesh"); manager.EnableGroup("LeftRigid"); } else { manager.DisableGroup("LeftMesh"); manager.DisableGroup("LeftRigid"); } if (rightHandActiveFlag) { manager.EnableGroup("RightMesh"); manager.EnableGroup("RightRigid"); } else { manager.DisableGroup("RightMesh"); manager.DisableGroup("RightRigid"); } }

additionalFrameSave()

protected override void additionalFrameSave(BinaryWriter binWriter) { if (saveBoneColliderShapes) { foreach (Collider c in allCollidersInHands) { if (c is CapsuleCollider cap) { binWriter.Write(1.0f); // capsule binWriter.Write(cap.center.x); binWriter.Write(cap.center.y); binWriter.Write(cap.center.z); binWriter.Write(cap.height); binWriter.Write(cap.radius); binWriter.Write((float)cap.direction); } else if (c is BoxCollider box) { binWriter.Write(2.0f); // box binWriter.Write(box.center.x); binWriter.Write(box.center.y); binWriter.Write(box.center.z); binWriter.Write(box.size.x); binWriter.Write(box.size.y); binWriter.Write(box.size.z); } } } }

additionalInfoSave()

protected override void additionalInfoSave(BinaryWriter binWriter) { // Reserved for extension }

applyLeftPerturbation()

public override void applyLeftPerturbation(Vector3 offset) { leftHandOffset = offset; leftHandMeshScript.perturbationOffset = offset; leftHandColliderScript.perturbationOffset = offset; for (int i = 0; i < 5; i++) { leftFingersMeshScripts[i].perturbationOffset = offset; leftFingersColliderScripts[i].perturbationOffset = offset; } }

applyRightPerturbation()

public override void applyRightPerturbation(Vector3 offset) { rightHandOffset = offset; rightHandMeshScript.perturbationOffset = offset; rightHandColliderScript.perturbationOffset = offset; for (int i = 0; i < 5; i++) { rightFingersMeshScripts[i].perturbationOffset = offset; rightFingersColliderScripts[i].perturbationOffset = offset; } }
