Documentation: questHandsAndSaveScript.cs

1. Overview

questHandsAndSaveScript is a subclass of handsAndSaveScript designed specifically for Oculus Quest hand tracking using Unity's XR Hands framework. It handles hand mesh activation, tracking detection, fingertip and collider data saving, and gain perturbation offsets through offsetHandProcessor.

2. Functional Blocks

Rig Initialization

  • rigSetup() links Unity XR hand and collider objects, headset, wrist positions, and fingertip markers if saving is enabled.

Tracking Status

  • leftNotTracking() and rightNotTracking() determine whether the respective hands are being actively tracked.

Data Saving

  • additionalFrameSave(): Writes fingertip positions if enabled.
  • additionalInfoSave(): Writes shape parameters for hand colliders if enabled.

Hand State Management

  • setHandGameobjectActivity(): Activates or disables each hand’s mesh and colliders based on current task flags.

Perturbation Control

  • applyLeftPerturbation() and applyRightPerturbation(): Apply real-time gain perturbations via offsetHandProcessor.

rigSetup()

protected override void rigSetup() { headsetTransform = transform.Find("/XR Origin/Camera Offset/Main Camera"); leftHandCollidersObject = transform.Find("/XR Origin/Camera Offset/Left Hand Colliders").gameObject; rightHandCollidersObject = transform.Find("/XR Origin/Camera Offset/Right Hand Colliders").gameObject; leftHandTrackingObject = transform.Find("/XR Origin/Camera Offset/Left Hand Tracking").gameObject; rightHandTrackingObject = transform.Find("/XR Origin/Camera Offset/Right Hand Tracking").gameObject; leftHandTracker = leftHandTrackingObject.transform.Find("LeftHand").GetComponent<SkinnedMeshRenderer>(); rightHandTracker = rightHandTrackingObject.transform.Find("RightHand").GetComponent<SkinnedMeshRenderer>(); leftWrist = transform.Find("/XR Origin/Camera Offset/Left Hand Tracking/L_Wrist"); rightWrist = transform.Find("/XR Origin/Camera Offset/Right Hand Tracking/R_Wrist"); handProcessorScript = transform.Find("/XR Origin/Camera Offset/Hand Visualizer") .GetComponent<offsetHandProcessor>(); allCollidersLeftHand = leftHandCollidersObject.GetComponentsInChildren<CapsuleCollider>(true); allCollidersRightHand = rightHandCollidersObject.GetComponentsInChildren<CapsuleCollider>(true); allCollidersInHands = allCollidersLeftHand.Concat(allCollidersRightHand).ToArray(); if (saveFingertipPosition) { var allMarkers = leftHandCollidersObject.GetComponentsInChildren<Transform>(true) .Concat(rightHandCollidersObject.GetComponentsInChildren<Transform>(true)); int i = 0; foreach (var marker in allMarkers) { if (marker.name.Contains("Tip")) allFingertips[i++] = marker; } } }

Tracking Status Checks

protected override bool leftNotTracking() { return !leftHandTrackingObject.activeInHierarchy || !leftHandTracker.enabled; } protected override bool rightNotTracking() { return !rightHandTrackingObject.activeInHierarchy || !rightHandTracker.enabled; }

additionalFrameSave()

protected override void additionalFrameSave(BinaryWriter binWriter) { if (saveFingertipPosition) { foreach (Transform fingertip in allFingertips) { binWriter.Write(fingertip.position.x); binWriter.Write(fingertip.position.y); binWriter.Write(fingertip.position.z); } } }

additionalInfoSave()

protected override void additionalInfoSave(BinaryWriter binWriter) { if (saveBoneColliderShapes) { foreach (CapsuleCollider c in allCollidersInHands) { binWriter.Write(c.center.x); binWriter.Write(c.center.y); binWriter.Write(c.center.z); binWriter.Write(c.height); binWriter.Write(c.radius); binWriter.Write((float)c.direction); } } }

setHandGameobjectActivity()

protected override void setHandGameobjectActivity() { if (leftHandActiveFlag) { leftHandTrackingObject.SetActive(true); leftHandCollidersObject.GetComponent<handBonesScript>().resetToActive(); } else { leftHandTrackingObject.SetActive(false); leftHandCollidersObject.SetActive(false); } if (rightHandActiveFlag) { rightHandTrackingObject.SetActive(true); rightHandCollidersObject.GetComponent<handBonesScript>().resetToActive(); } else { rightHandTrackingObject.SetActive(false); rightHandCollidersObject.SetActive(false); } }

applyLeftPerturbation() & applyRightPerturbation()

public override void applyLeftPerturbation(Vector3 offset) { leftHandOffset = offset; handProcessorScript.leftHandOffset = offset; } public override void applyRightPerturbation(Vector3 offset) { rightHandOffset = offset; handProcessorScript.rightHandOffset = offset; }
