Documentation: plateAndGrapeScript.cs

1. Overview

The plateAndGrapeScript class manages the visual, physical, and feedback behavior of the plate and grape objects during each trial. It handles trial stage transitions (e.g., setup, interaction, and deactivation), resets transforms, controls feedback particle effects, and toggles the physical state of the objects based on collision or tracking conditions.

2. Functional Blocks

Initialization

  • startup(): Finds key child objects (plate, grape), gets component references, and prepares internal state.
  • storeInitialPlateTransform(): Stores initial position and rotation of plate and grape for future reset.

Trial Stage Handling

  • changeTrialStage(): Responds to trial stages to reset, activate, disable, or provide feedback through particle effects.

Physics and Effects

  • resetAndActivate(): Repositions and activates plate or grape with cleared velocity.
  • setPlateKinematic(): Freezes or unfreezes movement of plate and grape using rigidbody constraints.
  • getPlatePenetration(): Detects collisions between the plate and tracked hand colliders using Unity's ComputePenetration().

Particle Effects

  • playEffect(): Configures and plays the feedback particle effect based on trial outcome.
  • getEffectColor(): Selects color for the particle effect based on success or failure.

Internal Fields

startup()

public void startup() { particle = GetComponent<ParticleSystem>(); plateScriptVar = GetComponentInChildren<plateScript>(true); plate = transform.Find("Plate").gameObject; grape = transform.Find("Grape").gameObject; }

changeTrialStage()

public void changeTrialStage(int stage, float info) { switch (stage) { case 5: // End of reach trialSuccess = info > 0; playEffect(); grape.SetActive(false); plate.SetActive(false); break; case 3: // Before reach begins resetAndActivate(plate, plateStartingPosition, plateStartingRotation); resetAndActivate(grape, grapeStartingPosition, grapeStartingRotation); playEffect(); plateScriptVar.waitingForHandContactFlag = true; break; case 4: // Reach has begun plateScriptVar.waitingForHandContactFlag = false; break; } }

resetAndActivate()

void resetAndActivate(GameObject obj, Vector3 pos, Quaternion rot) { Rigidbody rb = obj.GetComponent<Rigidbody>(); rb.velocity = Vector3.zero; rb.angularVelocity = Vector3.zero; obj.transform.localPosition = pos; obj.transform.rotation = rot; obj.SetActive(true); }

playEffect() and getEffectColor()

void playEffect() { var shape = particle.shape; shape.position = plate.transform.localPosition; var main = particle.main; main.startColor = getEffectColor(); particle.Play(); } Color getEffectColor() { return trialSuccess ? new Color(plateGrapeColorSuccessR, plateGrapeColorSuccessG, plateGrapeColorSuccessB, plateGrapeColorSuccessAlpha) : new Color(plateGrapeColorUnsuccessR, plateGrapeColorUnsuccessG, plateGrapeColorUnsuccessB, plateGrapeColorUnsuccessAlpha); }

storeInitialPlateTransform()

public void storeInitialPlateTransform() { plateStartingPosition = plate.transform.localPosition; plateStartingRotation = plate.transform.rotation; grapeStartingPosition = grape.transform.localPosition; grapeStartingRotation = grape.transform.rotation; }

setPlateKinematic()

public void setPlateKinematic(bool kineFlag) { Rigidbody plateRb = plate.GetComponent<Rigidbody>(); Rigidbody grapeRb = grape.GetComponent<Rigidbody>(); if (kineFlag) { plateRb.constraints = RigidbodyConstraints.FreezeAll; grapeRb.constraints = RigidbodyConstraints.FreezeAll; } else { plateRb.constraints = RigidbodyConstraints.None; grapeRb.constraints = RigidbodyConstraints.None; } }

getPlatePenetration()

public bool getPlatePenetration(Collider[] inputObjectColliders) { Collider[] plateColliders = plate.GetComponentsInChildren<Collider>(); foreach (Collider plateCollider in plateColliders) { if (!plateCollider.isTrigger) { foreach (Collider inputCollider in inputObjectColliders) { if (Physics.ComputePenetration( plateCollider, plateCollider.transform.position, plateCollider.transform.rotation, inputCollider, inputCollider.transform.position, inputCollider.transform.rotation, out Vector3 direction, out float distance)) { return true; } } } } return false; }
